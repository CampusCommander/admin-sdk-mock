# Admin SDK Mock Milestone 2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the milestone 2 writes for Chrome devices, device commands, users, and OUs, with lagging lists, write limits, and write latency.

**Architecture:** Writes are new routes in the existing dispatch pipeline. Every write saves before-images of the rows it changes. List handlers read through one SQLite view per resource. The view substitutes each before-image until the search delay passes. A device command stores its whole timeline when it is issued. Its side effects apply lazily, stamped at the execution time.

**Tech Stack:** Node 24 (type stripping, `node:test`, `node:sqlite`), TypeScript (type checks only), ESLint, Prettier. Compatibility tests use `@googleapis/admin` and `google-auth-library`. No new dependencies.

**Spec:** `docs/specs/2026-10-09-milestone-2-writes-design.md`. Its parent is `docs/specs/2026-10-09-admin-sdk-mock-design.md`. Read both before any task.

## Global Constraints

- Repository: `~/projects/admin-sdk-mock`. Work on branch `codex/milestone-2` from `main`. Never add files to `~/projects/campus-commander`.
- Node runs `.ts` files directly through type stripping. Never use `enum`, `namespace`, parameter properties, decorators, or `import x = require()`.

- Add no runtime or development dependency.
- Directory errors use `directoryError()` from `src/http/errors.ts`. Never build an error envelope by hand.
- Write routes require the read-write scope of their resource and a super administrator subject.
- A write stamps its before-images with `call.now`. The search delay in milliseconds is `settings.writes.searchDelaySeconds * 1000 / call.scale`. A delay of 0 saves no before-images.
- The time scale divides every duration: the search delay, the command timeline, the command expiry, the undelete window, and the write-limit windows.
- **Ruling:** the 24-hour online window for device commands compares seeded timestamps. It does not scale.
- Write responses without a response schema return `204` with an empty body.

```ts
const invalidJson = () =>
  new ApiError(
    directoryError(400, 'invalid', 'Invalid JSON payload received.', {
      extra: { status: 'INVALID_ARGUMENT' },
    }),
  );

/** Parse a JSON object body. An empty body reads as an empty object. */
export function jsonBody(call: Call): Record<string, unknown> {
  const text = call.request.body.trim();
  if (text === '') return {};
  let value: unknown;
  try {
    value = JSON.parse(text);
  } catch {
    throw invalidJson();
  }
  if (value === null || typeof value !== 'object' || Array.isArray(value))
    throw invalidJson();
  return value as Record<string, unknown>;
}

/** Read an optional string field. `null` reads as null. `max` counts characters. */
export function stringField(
  call: Call,
  body: Record<string, unknown>,
  name: string,
  max?: number,
): string | null | undefined {
  if (!Object.hasOwn(body, name)) return undefined;
  const value = body[name];
  if (value === null) return null;
  if (typeof value !== 'string')
    throw invalidArgument(call, `Invalid Input: ${name}`);
  if (max !== undefined && [...value].length > max)
    throw invalidArgument(
      call,
      `Invalid Input: ${name} is longer than ${max} characters.`,
    );
  return value;
}

/** Read an optional boolean field. `null` reads as false. */
export function booleanField(
  call: Call,
  body: Record<string, unknown>,
  name: string,
): boolean | undefined {
  if (!Object.hasOwn(body, name)) return undefined;
  const value = body[name];
  if (value === null) return false;
  if (typeof value !== 'boolean')
    throw invalidArgument(call, `Invalid Input: ${name}`);
  return value;
}

export const failedPrecondition = (message: string): ApiError =>
  new ApiError(
    directoryError(400, 'failedPrecondition', message, {
      extra: { status: 'FAILED_PRECONDITION' },
    }),
  );

export const duplicate = (): ApiError =>
  new ApiError(directoryError(409, 'duplicate', 'Entity already exists.'));

export const notFound = (what: string): ApiError =>
  new ApiError(directoryError(404, 'notFound', `Resource Not Found: ${what}`));

/** The search delay of the call's domain in scaled milliseconds. */
export function searchDelayMs(call: Call): number {
  return (
    (domainOf(call).record.settings.writes.searchDelaySeconds * 1000) /
    call.scale
  );
}
```

```ts
const T0 = Date.UTC(2026, 9, 9, 12);
const D = 300_000;

function seeded(t: import('node:test').TestContext) {
  const dir = mkdtempSync(join(tmpdir(), 'lag-'));
  const db = openDomainDb(join(dir, 'domain.sqlite'));
  t.after(() => {
    db.close();
    rmSync(dir, { recursive: true, force: true });

  seedDomain(
    db,

      seed: 'lag',
```

```ts
import type { DatabaseSync } from 'node:sqlite';
import { row, rows } from './sqlite.ts';

/** Each table whose lists lag writes, and its key column. */
export const LAGGED = {
  orgunits: 'org_unit_id',
  users: 'id',
  deleted_users: 'id',
  devices: 'device_id',
} as const;
export type LaggedTable = keyof typeof LAGGED;

/** Before-image indexes that keep the list sorts on indexes. */
const PRIOR_INDEXES: Readonly<Record<LaggedTable, readonly string[]>> = {
  orgunits: ['parent_id, ord', 'path'],
  users: ['primary_email', 'ord'],
  deleted_users: ['primary_email'],
  devices: ['org_unit_path, ord', 'ord'],
};

const TABLES = Object.keys(LAGGED) as LaggedTable[];
const columnCache = new WeakMap<DatabaseSync, Map<string, string[]>>();

/** The column names of a table in declaration order. */
export function columnsOf(db: DatabaseSync, table: string): string[] {
  let cache = columnCache.get(db);
  if (!cache) {
    cache = new Map();
    columnCache.set(db, cache);
  }
  let columns = cache.get(table);
  if (!columns) {
    columns = rows<{ name: string }>(
      db.prepare(`PRAGMA table_info(${table})`).all(),
    ).map((column) => column.name);
    cache.set(table, columns);
  }
  return columns;
}

/**
 * Create one before-image table and one listed view per lagged table.
 * The listed view shows the earliest waiting before-image of a row, or the live row.
 */
export function createLagSchema(db: DatabaseSync): void {
  for (const table of TABLES) {
    const key = LAGGED[table];
    const prior = `${table}_prior`;
    const columns = columnsOf(db, table);
    db.exec(
      `CREATE TABLE IF NOT EXISTS ${prior} (seq INTEGER PRIMARY KEY, written_at INTEGER NOT NULL, absent INTEGER NOT NULL, ${columns.join(', ')})`,
    );
    const existing = new Set(
      rows<{ name: string }>(
        db.prepare(`PRAGMA table_info(${prior})`).all(),
      ).map((column) => column.name),
    );
    for (const column of columns)
      if (!existing.has(column))
        db.exec(`ALTER TABLE ${prior} ADD COLUMN ${column}`);
    db.exec(
      `CREATE INDEX IF NOT EXISTS ${prior}_key ON ${prior} (${key}, seq)`,
    );
    db.exec(
      `CREATE INDEX IF NOT EXISTS ${prior}_written ON ${prior} (written_at)`,
    );
    PRIOR_INDEXES[table].forEach((index, n) =>
      db.exec(`CREATE INDEX IF NOT EXISTS ${prior}_x${n} ON ${prior} (${index})`),
    );
    const list = columns.join(', ');
    db.exec(`DROP VIEW IF EXISTS ${table}_listed`);
    db.exec(
      `CREATE VIEW ${table}_listed AS ` +
        `SELECT ${list} FROM ${table} live WHERE NOT EXISTS (SELECT 1 FROM ${prior} p WHERE p.${key} = live.${key}) ` +
        `UNION ALL ` +
        `SELECT ${list} FROM ${prior} p WHERE p.absent = 0 AND p.seq = (SELECT MIN(q.seq) FROM ${prior} q WHERE q.${key} = p.${key})`,
    );
  }
}

/** Save the live row as a before-image, or an absent marker when no row exists. */
export function saveBeforeImage(
  db: DatabaseSync,
  table: LaggedTable,
  keyValue: string,
  writtenAt: number,
  delayMs: number,
): void {
  if (delayMs <= 0) return;
  const key = LAGGED[table];
  const list = columnsOf(db, table).join(', ');
  const saved = db
    .prepare(
      `INSERT INTO ${table}_prior (written_at, absent, ${list}) SELECT ?, 0, ${list} FROM ${table} WHERE ${key} = ?`,
    )
    .run(writtenAt, keyValue);
  if (Number(saved.changes) === 0)
    db.prepare(
      `INSERT INTO ${table}_prior (written_at, absent, ${key}) VALUES (?, 1, ?)`,
    ).run(writtenAt, keyValue);
}

/** Save a before-image of every live row that matches the condition. */
export function saveBeforeImagesWhere(
  db: DatabaseSync,
  table: LaggedTable,
  where: string,
  params: readonly (string | number)[],
  writtenAt: number,
  delayMs: number,
): number {
  if (delayMs <= 0) return 0;
  const list = columnsOf(db, table).join(', ');
  return Number(
    db
      .prepare(
        `INSERT INTO ${table}_prior (written_at, absent, ${list}) SELECT ?, 0, ${list} FROM ${table} WHERE ${where}`,
      )
      .run(writtenAt, ...params).changes,
  );
}

/** Delete the before-images that the delay has passed, then name the relation that lists read. */
export function listedRelation(
  db: DatabaseSync,
  table: LaggedTable,
  now: number,
  delayMs: number,
): string {
  db.prepare(`DELETE FROM ${table}_prior WHERE written_at <= ?`).run(
    now - delayMs,
  );
  return db.prepare(`SELECT 1 FROM ${table}_prior LIMIT 1`).get()
    ? `${table}_listed`
    : table;
}

/** Delete every before-image. Returns the number deleted. */
export function flushLag(db: DatabaseSync): number {
  let total = 0;
  for (const table of TABLES)
    total += Number(db.prepare(`DELETE FROM ${table}_prior`).run().changes);
  return total;
}

/** Count the before-images that still wait for the delay. */
export function pendingLag(
  db: DatabaseSync,
  now: number,
  delayMs: number,
): number {
  let total = 0;
  for (const table of TABLES)
    total += Number(
      row<{ n: number }>(
        db
          .prepare(
            `SELECT count(*) AS n FROM ${table}_prior WHERE written_at > ?`,
          )
          .get(now - delayMs),
      )?.n ?? 0,
    );
  return total;
}
```

```ts
export interface BatchRequestPart {
  id: string;
  method: string;
  /** An absolute Directory path, such as /admin/directory/v1/users. */
  path: string;
  body?: unknown;
}

/** Build a multipart/mixed batch body. */
export function batchBody(
  parts: readonly BatchRequestPart[],
  boundary = 'batch',
): string {
  return (
    parts
      .map((part) => {
        const head = `--${boundary}\r\nContent-Type: application/http\r\nContent-ID: <${part.id}>\r\n\r\n${part.method} ${part.path}\r\n`;
        return part.body === undefined
          ? `${head}\r\n`
          : `${head}Content-Type: application/json\r\n\r\n${JSON.stringify(part.body)}\r\n`;
      })
      .join('') + `--${boundary}--\r\n`
  );
}

/** The HTTP status of each part of a batch response, by request Content-ID. */
export function batchStatuses(text: string): Map<string, number> {
  const statuses = new Map<string, number>();
  for (const match of text.matchAll(
    /Content-ID: <response-([^>]+)>\r\n\r\nHTTP\/1\.1 (\d{3})/g,
  ))
    statuses.set(match[1], Number(match[2]));
  return statuses;
}
```

```ts
import { ApiError, directoryError } from '../http/errors.ts';
import type { ApiResponse, Call } from '../http/types.ts';
import type { StoredDevice } from '../seed/devices.ts';
import { saveBeforeImage, saveBeforeImagesWhere } from '../store/lag.ts';
import { row, rows, transaction } from '../store/sqlite.ts';
import {
  domainOf,
  enumParam,
  invalidArgument,
  jsonBody,
  resolveCustomer,
  searchDelayMs,
  stringField,
} from './common.ts';
import { deviceResource, visibleDevice } from './devices.ts';
import { targetOrgUnit } from './orgunits.ts';

export const MAX_DEVICE_IDS = 50;
const PROJECTION = { basic: 'BASIC', full: 'FULL' } as const;

/** Writable annotation fields, their columns, and their character limits. */
const ANNOTATIONS: readonly { field: string; column: string; max?: number }[] =
  [
    { field: 'annotatedUser', column: 'annotated_user', max: 100 },
    { field: 'annotatedLocation', column: 'annotated_location', max: 200 },
    { field: 'annotatedAssetId', column: 'annotated_asset_id' },
    { field: 'notes', column: 'notes', max: 500 },
  ];

/** Read `deviceIds`: 1 to 50 non-empty strings, in request order. */
export function deviceIdList(call: Call, value: unknown): string[] {
  if (
    !Array.isArray(value) ||
    value.length === 0 ||
    !value.every((id) => typeof id === 'string' && id !== '')
  )
    throw invalidArgument(call, 'Invalid Input: deviceIds');
  if (value.length > MAX_DEVICE_IDS)
    throw invalidArgument(
      call,
      `Invalid Input: deviceIds holds more than ${MAX_DEVICE_IDS} devices.`,
    );
  return value as string[];
}

/** `chromeosdevices.update` and `chromeosdevices.patch`. Both change only the fields in the body. */
export function updateDevice(call: Call): ApiResponse {
  const domain = domainOf(call);
  resolveCustomer(call, call.params.customerId);
  const projection = enumParam(call, 'projection', PROJECTION, 'FULL');
  const device = visibleDevice(call, call.params.deviceId);
  const body = jsonBody(call);
  const sets: string[] = [];
  const params: (string | null)[] = [];
  for (const { field, column, max } of ANNOTATIONS) {
    const value = stringField(call, body, field, max);
    if (value === undefined) continue;
    sets.push(`${column} = ?`);
    params.push(value === null || value === '' ? null : value);
  }
  const ouPath = stringField(call, body, 'orgUnitPath');
  const ouId = stringField(call, body, 'orgUnitId');
  const ouRaw = ouPath ?? (ouId ? (ouId.startsWith('id:') ? ouId : `id:${ouId}`) : null);
  if (ouRaw) {
    const unit = targetOrgUnit(call, ouRaw);
    sets.push('org_unit_path = ?', 'org_unit_id = ?');
    params.push(unit.path, unit.org_unit_id);
  }
  if (sets.length > 0)
    transaction(domain.db, () => {
      saveBeforeImage(
        domain.db,
        'devices',
        device.device_id,
        call.now,
        searchDelayMs(call),
      );
      domain.db
        .prepare(`UPDATE devices SET ${sets.join(', ')} WHERE device_id = ?`)
        .run(...params, device.device_id);
    });
  const updated = row<StoredDevice>(
    domain.db
      .prepare('SELECT * FROM devices WHERE device_id = ?')
      .get(device.device_id),
  )!;
  return {
    status: 200,
    items: 0,
    body: deviceResource(updated, projection, domain),
  };
}

/** `chromeosdevices.moveDevicesToOu`. One unknown device fails the whole request. */
export function moveDevicesToOu(call: Call): ApiResponse {
  const domain = domainOf(call);
  resolveCustomer(call, call.params.customerId);
  const raw = call.request.query.get('orgUnitPath');
  if (!raw)
    throw new ApiError(
      directoryError(400, 'required', 'Required parameter: orgUnitPath'),
    );
  const unit = targetOrgUnit(call, raw);
  const ids = [...new Set(deviceIdList(call, jsonBody(call).deviceIds))];
  const marks = ids.map(() => '?').join(', ');
  const hidden = call.overrides.hidden(domain.customerId);
  const known = new Set(
    rows<{ device_id: string }>(
      domain.db
        .prepare(`SELECT device_id FROM devices WHERE device_id IN (${marks})`)
        .all(...ids),
    ).map((found) => found.device_id),
  );
  const unknown = ids.find((id) => !known.has(id) || hidden.has(id));
  if (unknown !== undefined)
    throw invalidArgument(call, `Invalid Input: deviceId ${unknown}`);
  transaction(domain.db, () => {
    saveBeforeImagesWhere(
      domain.db,
      'devices',
      `device_id IN (${marks})`,
      ids,
      call.now,
      searchDelayMs(call),
    );
    domain.db
      .prepare(
        `UPDATE devices SET org_unit_path = ?, org_unit_id = ? WHERE device_id IN (${marks})`,
      )
      .run(unit.path, unit.org_unit_id, ...ids);
  });
  return { status: 204, items: ids.length };
}
```

```ts
  'DEPROVISION_REASON_DOMAIN_MOVE',

  'DEPROVISION_REASON_UPGRADE_TRANSFER',
  'DEPROVISION_REASON_NOT_REQUIRED',
  'DEPROVISION_REASON_REPAIR_CENTER',

/** `customer.devices.chromeos.batchChangeStatus`. Each device succeeds or fails alone. */
export function batchChangeStatus(call: Call): ApiResponse {
  const domain = domainOf(call);
  resolveCustomer(call, call.params.customerId);
  const body = jsonBody(call);
  const ids = deviceIdList(call, body.deviceIds);
  const name = body.changeChromeOsDeviceStatusAction;
  const action =
    typeof name === 'string' && Object.hasOwn(ACTIONS, name)
      ? ACTIONS[name]
      : undefined;
  if (!action)
    throw invalidArgument(
      call,
      'Invalid Input: changeChromeOsDeviceStatusAction',
    );
  const reason = body.deprovisionReason;
  const deprovision = action.to === 'DEPROVISIONED';
  if (deprovision && (typeof reason !== 'string' || !REASONS.has(reason)))
    throw invalidArgument(call, 'Invalid Input: deprovisionReason');
  const hidden = call.overrides.hidden(domain.customerId);
  const select = domain.db.prepare(
    'SELECT status FROM devices WHERE device_id = ?',
  );
  const delay = searchDelayMs(call);
  const results = transaction(domain.db, () =>
    ids.map((deviceId) => {
      const found = hidden.has(deviceId)
        ? undefined
        : row<{ status: string }>(select.get(deviceId));
      if (!found)
        return {
          deviceId,
          error: { code: NOT_FOUND, message: 'Device not found.' },
        };
      if (!action.from.includes(found.status))
        return {
          deviceId,
          error: {
            code: FAILED_PRECONDITION,
            message: `A device with status ${found.status} does not allow this action.`,
          },
        };
      saveBeforeImage(domain.db, 'devices', deviceId, call.now, delay);
      if (deprovision)
        domain.db
          .prepare(
            'UPDATE devices SET status = ?, deprovision_reason = ?, last_deprovision = ? WHERE device_id = ?',
          )
          .run(action.to, reason as string, call.now, deviceId);
      else
        domain.db
          .prepare('UPDATE devices SET status = ? WHERE device_id = ?')
          .run(action.to, deviceId);
      return { deviceId, response: {} };
    }),
  );
  return {
    status: 200,
    items: ids.length,
    body: { changeChromeOsDeviceStatusResults: results },
  };
}
```

```ts
import type { Call } from '../http/types.ts';
import { saveBeforeImage } from '../store/lag.ts';
import { rows, transaction } from '../store/sqlite.ts';
import { domainOf, searchDelayMs } from './common.ts';

/** One row of the `device_commands` table. */
export interface CommandRow {
  command_id: number;
  device_id: string;
  type: string;
  payload: string | null;
  issued_at: number;
  sent_at: number | null;
  acked_at: number | null;
  executed_at: number | null;
  expire_at: number;
  result: string;
  error_message: string | null;
  applied: number;
}

/** A command executes only when its execution time comes before its expiry. */
const executes = (command: CommandRow): boolean =>
  command.executed_at !== null && command.executed_at <= command.expire_at;

/** The state of a command at a time. */
export function commandState(command: CommandRow, now: number): string {
  if (executes(command) && now >= command.executed_at!)
    return 'EXECUTED_BY_CLIENT';
  if (now >= command.expire_at) return 'EXPIRED';
  if (command.acked_at !== null && now >= command.acked_at)
    return 'ACKED_BY_CLIENT';
  if (command.sent_at !== null && now >= command.sent_at)
    return 'SENT_TO_CLIENT';
  return 'PENDING';
}

const EFFECTS: Readonly<
  Record<string, (command: CommandRow) => [string, (string | number)[]]>
> = {
  REBOOT: (command) => [
    'UPDATE devices SET last_sync = ? WHERE device_id = ?',
    [command.executed_at!, command.device_id],
  ],
  WIPE_USERS: (command) => [
    'UPDATE devices SET recent_users = NULL WHERE device_id = ?',
    [command.device_id],
  ],
  REMOTE_POWERWASH: (command) => [
    'UPDATE devices SET recent_users = NULL WHERE device_id = ?',
    [command.device_id],
  ],
};

/**
 * Apply the side effects of every successful command that has executed.
 * Each side effect is a write stamped at its execution time.
 * Run this before any handler reads or writes device rows.
 */
export function settleCommands(call: Call): void {
  const domain = domainOf(call);
  const due = rows<CommandRow>(
    domain.db
      .prepare(
        'SELECT * FROM device_commands WHERE applied = 0 AND executed_at IS NOT NULL AND executed_at <= ? AND executed_at <= expire_at ORDER BY executed_at, command_id',
      )
      .all(call.now),
  );
  if (due.length === 0) return;
  const delay = searchDelayMs(call);
  transaction(domain.db, () => {
    for (const command of due) {
      const effect = EFFECTS[command.type];
      if (effect && command.result === 'SUCCESS') {
        saveBeforeImage(
          domain.db,
          'devices',
          command.device_id,
          command.executed_at!,
          delay,
        );
        const [sql, params] = effect(command);
        domain.db.prepare(sql).run(...params);
      }
      domain.db
        .prepare('UPDATE device_commands SET applied = 1 WHERE command_id = ?')
        .run(command.command_id);
    }
  });
}
```

```ts
import { visibleDevice } from './devices.ts';

export const COMMAND_TYPES: ReadonlySet<string> = new Set([
  'REBOOT',
  'TAKE_A_SCREENSHOT',
  'SET_VOLUME',
  'WIPE_USERS',
  'REMOTE_POWERWASH',
  'DEVICE_START_CRD_SESSION',
  'CAPTURE_LOGS',
  'FETCH_CRD_AVAILABILITY_INFO',
  'FETCH_SUPPORT_PACKET',
]);

/** Ruling: a device synced within 24 hours is online. The window compares seeded timestamps, so it does not scale. */
const ONLINE_MS = 24 * 3_600_000;
const SUCCESS_RATE = 0.95;
const FAILURE_MESSAGE = 'The device failed to execute the command.';
const COMMAND_ID_BASE = 4_100_000_000;

function parseObject(text: string): Record<string, unknown> | undefined {
  try {
    const value: unknown = JSON.parse(text);
    return value !== null && typeof value === 'object' && !Array.isArray(value)
      ? (value as Record<string, unknown>)
      : undefined;
  } catch {
    return undefined;
  }
}

/** Validate the payload of a command. Only SET_VOLUME and DEVICE_START_CRD_SESSION take one. */
function payloadFor(call: Call, type: string, raw: unknown): string | null {
  const invalid = () => invalidArgument(call, 'Invalid Input: payload');
  if (raw === undefined || raw === null || raw === '') {
    if (type === 'SET_VOLUME') throw invalid();
    return null;
  }
  if (typeof raw !== 'string') throw invalid();
  if (type === 'SET_VOLUME') {
    const volume = parseObject(raw)?.volume;
    if (
      typeof volume !== 'number' ||
      !Number.isInteger(volume) ||
      volume < 0 ||
      volume > 100
    )
      throw invalid();
    return raw;
  }
  if (type === 'DEVICE_START_CRD_SESSION') {
    if (!parseObject(raw)) throw invalid();
    return raw;
  }
  throw invalidArgument(call, `Invalid Input: ${type} takes no payload.`);
}

/** `customer.devices.chromeos.issueCommand`. The response holds the new command ID. */
export function issueCommand(call: Call): ApiResponse {
  const domain = domainOf(call);
  resolveCustomer(call, call.params.customerId);
  settleCommands(call);
  const device = visibleDevice(call, call.params.deviceId);
  const body = jsonBody(call);
  const type = body.commandType;
  if (typeof type !== 'string' || !COMMAND_TYPES.has(type))
    throw invalidArgument(call, 'Invalid Input: commandType');
  const payload = payloadFor(call, type, body.payload);
  if (device.status !== 'ACTIVE')
    throw failedPrecondition('The device must be active to receive a command.');
  const commandId = Number(
    row<{ next: number }>(
      domain.db
        .prepare(
          'SELECT COALESCE(MAX(command_id), ?) + 1 AS next FROM device_commands',
        )
        .get(COMMAND_ID_BASE),
```

```ts
const EMAIL = /^[^@\s]+@[^@\s]+$/;

/** Read a primary email on one of the customer's own domains. Alias domains do not count. */
export function primaryEmail(call: Call, value: unknown): string {
  if (typeof value !== 'string' || !EMAIL.test(value.trim()))
    throw invalidArgument(call, 'Invalid Input: primaryEmail');
  const email = value.trim().toLowerCase();
  const own = row(
    domainOf(call)

        'SELECT 1 FROM domains WHERE domain_name = ? AND alias_of IS NULL',
      )
      .get(email.slice(email.indexOf('@') + 1)),
  );
  if (!own) throw invalidArgument(call, 'Invalid Input: primaryEmail');
  return email;
}

export function emailInUse(
  db: DatabaseSync,
  email: string,
  exceptUserId = '',
): boolean {
  return (
    db
      .prepare(
        'SELECT 1 FROM users WHERE primary_email = ? AND id != ? UNION ALL SELECT 1 FROM user_aliases WHERE alias = ? AND user_id != ? LIMIT 1',
      )
      .get(email, exceptUserId, email, exceptUserId) !== undefined
  );
}

function namePart(
  call: Call,
  name: Record<string, unknown>,
  field: string,
): string {
  const value = name[field];
  if (
    typeof value !== 'string' ||
    value.trim() === '' ||
    [...value].length > MAX_NAME
  )
    throw invalidArgument(call, `Invalid Input: name.${field}`);
  return value.trim();
}

/** Read a `name` object. With `current`, a missing part keeps its current value. */
export function readName(
  call: Call,
  value: unknown,
  current?: { given: string; family: string },
): { given: string; family: string } {
  if (value === null || typeof value !== 'object' || Array.isArray(value))
    throw invalidArgument(call, 'Invalid Input: name');
  const name = value as Record<string, unknown>;
  return {
    given:
      current && name.givenName === undefined
        ? current.given
        : namePart(call, name, 'givenName'),
    family:
      current && name.familyName === undefined
        ? current.family
        : namePart(call, name, 'familyName'),
  };
}

/** Check a password. Google never returns a password, so the mock never stores one. */
export function checkPassword(
  call: Call,
  body: Record<string, unknown>,
): void {
  const { password, hashFunction } = body;
  if (
    hashFunction !== undefined &&
    hashFunction !== null &&
    (typeof hashFunction !== 'string' || !HASH_FUNCTIONS.has(hashFunction))
  )
    throw invalidArgument(call, 'Invalid Input: hashFunction');
  if (typeof password !== 'string' || password === '')
    throw invalidArgument(call, 'Invalid Password');
  const length = [...password].length;
  if (!hashFunction && (length < 8 || length > 100))
    throw invalidArgument(call, 'Invalid Password');
}

/** Read `externalIds`. Null or an empty array clears. The first value stays searchable. */
export function readExternalIds(
  call: Call,
  value: unknown,
): { json: string | null; first: string | null } {
  if (value === null) return { json: null, first: null };
  if (
    !Array.isArray(value) ||
    !value.every(
      (entry) =>
        entry !== null &&
        typeof entry === 'object' &&
        typeof (entry as { value?: unknown }).value === 'string',
    )
  )
    throw invalidArgument(call, 'Invalid Input: externalIds');
  if (value.length === 0) return { json: null, first: null };
  return {
    json: JSON.stringify(value),
    first: (value[0] as { value: string }).value,
  };
}

function newUserId(call: Call, email: string): string {

  const random = createRandom(`user/${domain.spec.seed}/${email}/${call.now}`);
  const taken = domain.db.prepare(

  );

    if (taken.get(id, id) === undefined) return id;

/** `users.insert`. Ruling: a user created through the API has the staff role. */
export function insertUser(call: Call): ApiResponse {
  const domain = domainOf(call);
  const body = jsonBody(call);
  const email = primaryEmail(call, body.primaryEmail);
  const name = readName(call, body.name);
  checkPassword(call, body);
  const ou = stringField(call, body, 'orgUnitPath');
  const unit = ou
    ? targetOrgUnit(call, ou)
    : resolveOrgUnit(call, domain.db, '/');
  const suspended = booleanField(call, body, 'suspended') ?? false;
  const changePassword =
    booleanField(call, body, 'changePasswordAtNextLogin') ?? false;
  const external =
    body.externalIds === undefined
      ? { json: null, first: null }
      : readExternalIds(call, body.externalIds);
  if (emailInUse(domain.db, email)) throw duplicate();
  const id = newUserId(call, email);
  const ord = Number(
    row<{ next: number }>(
      domain.db
        .prepare(
          'SELECT COALESCE(MAX(ord), -1) + 1 AS next FROM (SELECT ord FROM users UNION ALL SELECT ord FROM deleted_users)',
        )
        .get(),
    )!.next,
  );
  transaction(domain.db, () => {
    saveBeforeImage(domain.db, 'users', id, call.now, searchDelayMs(call));
    domain.db
      .prepare(
        "INSERT INTO users (id, primary_email, given_name, family_name, full_name, org_unit_path, is_admin, suspended, suspension_reason, external_id, role, school, creation_time, last_login_time, ord, change_password_at_next_login, external_ids) VALUES (?, ?, ?, ?, ?, ?, 0, ?, ?, ?, 'staff', NULL, ?, NULL, ?, ?, ?)",
      )
      .run(
        id,
        email,
        name.given,
        name.family,
        `${name.given} ${name.family}`,
        unit.path,
        suspended ? 1 : 0,
        suspended ? 'ADMIN' : null,
        external.first,
        call.now,
        ord,
        changePassword ? 1 : 0,
        external.json,
      );
  });
  return { status: 200, body: writtenUser(call, id) };
}
```

```ts
  const key = call.params.userKey;
  const deleted = row<DeletedUserRow>(
    domain.db
      .prepare(
        'SELECT * FROM deleted_users WHERE (id = ? OR primary_email = ?) AND deleted_at > ? ORDER BY deleted_at DESC LIMIT 1',
      )
      .get(key, key.toLowerCase(), call.now - undeleteWindowMs(call)),
  );
  if (!deleted || call.overrides.hidden(domain.customerId).has(deleted.id)) {
    if (findUser(domain.db, key))
      throw invalidArgument(call, 'Invalid Input: the user is not deleted.');
    throw notFound('userKey');

    saveBeforeImage(domain.db, 'users', deleted.id, call.now, delay);

}
```

```ts
import type { DatabaseSync } from 'node:sqlite';
import type { ApiResponse, Call } from '../http/types.ts';
import { createRandom } from '../random.ts';
import { saveBeforeImage, saveBeforeImagesWhere } from '../store/lag.ts';
import { row, transaction } from '../store/sqlite.ts';
import { settleCommands } from './command-effects.ts';
import {
  domainOf,
  invalidArgument,
  jsonBody,
  resolveCustomer,
  searchDelayMs,
  stringField,
} from './common.ts';
import {
  orgUnitResource,
  resolveOrgUnit,
  targetOrgUnit,
  type OuRow,
} from './orgunits.ts';

/** Ruling: a name already used under the same parent answers 400. Names compare without case. */
const NAME_TAKEN =
  'Invalid Input: an org unit with this name already exists under the parent.';

function readName(call: Call, value: unknown): string {
  if (typeof value !== 'string' || value.trim() === '' || value.includes('/'))
    throw invalidArgument(call, 'Invalid Input: name');
  return value.trim();
}

/** The parent that the body names by path or by ID, if any. */
function namedParent(
  call: Call,
  body: Record<string, unknown>,
): OuRow | undefined {
  const path = stringField(call, body, 'parentOrgUnitPath');
  if (path) return targetOrgUnit(call, path);
  const id = stringField(call, body, 'parentOrgUnitId');
  if (id) return targetOrgUnit(call, id.startsWith('id:') ? id : `id:${id}`);
  return undefined;
}

const childPath = (parent: OuRow, name: string): string =>
  parent.path === '/' ? `/${name}` : `${parent.path}/${name}`;

function pathTaken(db: DatabaseSync, path: string, exceptId = ''): boolean {
  return (
    db
      .prepare(
        'SELECT 1 FROM orgunits WHERE lower(path) = lower(?) AND org_unit_id != ?',
      )
      .get(path, exceptId) !== undefined
  );
}

function newOrgUnitId(call: Call, path: string): string {
  const domain = domainOf(call);
  const random = createRandom(`ou/${domain.spec.seed}/${path}/${call.now}`);
  const exists = domain.db.prepare(
    'SELECT 1 FROM orgunits WHERE org_unit_id = ?',
  );
  for (;;) {
    const id = `id:0${random.string(13)}`;
    if (exists.get(id) === undefined) return id;
  }
}

function storedUnit(db: DatabaseSync, id: string): OuRow {
  return row<OuRow>(
    db.prepare('SELECT * FROM orgunits WHERE org_unit_id = ?').get(id),
  )!;
}

/**
 * Rewrite every path below `unit` from its old path to `newPath`.
 * Each changed OU, user, and device receives a before-image. Returns the rows changed.
 */
function cascade(
  db: DatabaseSync,
  unit: OuRow,
  newPath: string,
  newParent: OuRow,
  now: number,
  delay: number,
): number {
  const old = unit.path;
  const tail = old.length + 1;
  const tree = [old, tail, `${old}/`];
  const units = '(path = ? OR substr(path, 1, ?) = ?)';
  const members = '(org_unit_path = ? OR substr(org_unit_path, 1, ?) = ?)';
  saveBeforeImagesWhere(db, 'orgunits', units, tree, now, delay);
  saveBeforeImagesWhere(db, 'users', members, tree, now, delay);
  saveBeforeImagesWhere(db, 'devices', members, tree, now, delay);
  const depthDelta = newParent.depth + 1 - unit.depth;
  let changed = Number(
    db
      .prepare(
        `UPDATE orgunits SET path = ? || substr(path, ?), depth = depth + ?, parent_path = CASE WHEN org_unit_id = ? THEN parent_path ELSE ? || substr(parent_path, ?) END WHERE ${units}`,
      )
      .run(newPath, tail, depthDelta, unit.org_unit_id, newPath, tail, ...tree)
      .changes,
  );
  for (const table of ['users', 'devices'])
    changed += Number(
      db
        .prepare(
          `UPDATE ${table} SET org_unit_path = ? || substr(org_unit_path, ?) WHERE ${members}`,
        )
        .run(newPath, tail, ...tree).changes,
    );
  return changed;
}

/** `orgunits.insert`. */
export function insertOrgUnit(call: Call): ApiResponse {
  const domain = domainOf(call);
  resolveCustomer(call, call.params.customerId);
  const body = jsonBody(call);
  const name = readName(call, body.name);
  const parent = namedParent(call, body);
  if (!parent) throw invalidArgument(call, 'Invalid Input: parentOrgUnitPath');
  const path = childPath(parent, name);
  if (pathTaken(domain.db, path)) throw invalidArgument(call, NAME_TAKEN);
  const description = stringField(call, body, 'description') ?? '';
  const id = newOrgUnitId(call, path);
  const ord = Number(
    row<{ next: number }>(
      domain.db
        .prepare('SELECT COALESCE(MAX(ord), -1) + 1 AS next FROM orgunits')
        .get(),
    )!.next,
  );
  transaction(domain.db, () => {
    saveBeforeImage(domain.db, 'orgunits', id, call.now, searchDelayMs(call));
    domain.db
      .prepare(
        "INSERT INTO orgunits (org_unit_id, path, name, description, parent_id, parent_path, depth, kind, ord) VALUES (?, ?, ?, ?, ?, ?, ?, 'container', ?)",
      )
      .run(
        id,
        path,
        name,
        description,
        parent.org_unit_id,
        parent.path,
        parent.depth + 1,
        ord,
      );
  });
  return { status: 200, body: orgUnitResource(storedUnit(domain.db, id)) };
}

/** `orgunits.update` and `orgunits.patch`. Both change only the fields in the body. */
export function updateOrgUnit(call: Call): ApiResponse {
  const domain = domainOf(call);
  resolveCustomer(call, call.params.customerId);
  settleCommands(call);
  const unit = resolveOrgUnit(call, domain.db, call.params.orgUnitPath);
  const body = jsonBody(call);
  const rawName = stringField(call, body, 'name');
  const name =
    rawName === undefined || rawName === null
      ? unit.name
      : readName(call, rawName);
  const named = namedParent(call, body);
  const description = stringField(call, body, 'description');
  if (unit.parent_id === null && (name !== unit.name || named))
    throw invalidArgument(
      call,
      'Invalid Input: the root org unit cannot be renamed or moved.',
    );
  const parent =
    named ??
    (unit.parent_id === null
      ? undefined
      : storedUnit(domain.db, unit.parent_id));
  let path = unit.path;
  if (parent) {
    if (
      parent.org_unit_id === unit.org_unit_id ||
      parent.path.startsWith(`${unit.path}/`)
    )
      throw invalidArgument(
        call,
        'Invalid Input: an org unit cannot move into its own subtree.',
      );
    path = childPath(parent, name);
    if (path !== unit.path && pathTaken(domain.db, path, unit.org_unit_id))
      throw invalidArgument(call, NAME_TAKEN);
  }
  const delay = searchDelayMs(call);
  let cascaded = 0;
  transaction(domain.db, () => {
    if (parent && path !== unit.path)
      cascaded = cascade(domain.db, unit, path, parent, call.now, delay);
    else
      saveBeforeImage(
        domain.db,
        'orgunits',
        unit.org_unit_id,
        call.now,
        delay,
      );
    domain.db
      .prepare(
        'UPDATE orgunits SET name = ?, description = ?, parent_id = ?, parent_path = ? WHERE org_unit_id = ?',
      )
      .run(
        name,
        description ?? unit.description,
        parent?.org_unit_id ?? null,
        parent?.path ?? null,
        unit.org_unit_id,
      );
  });
  return {
    status: 200,
    scanned: cascaded,
    body: orgUnitResource(storedUnit(domain.db, unit.org_unit_id)),
  };
}

/** `orgunits.delete`. Only an empty OU other than the root can be deleted. */
export function deleteOrgUnit(call: Call): ApiResponse {
  const domain = domainOf(call);
  resolveCustomer(call, call.params.customerId);
  const unit = resolveOrgUnit(call, domain.db, call.params.orgUnitPath);
  if (unit.parent_id === null)
    throw invalidArgument(
      call,
      'Invalid Input: the root org unit cannot be deleted.',
    );
  const busy = domain.db
    .prepare(
      'SELECT 1 FROM orgunits WHERE parent_id = ? UNION ALL SELECT 1 FROM users WHERE org_unit_path = ? UNION ALL SELECT 1 FROM devices WHERE org_unit_path = ? LIMIT 1',
    )
    .get(unit.org_unit_id, unit.path, unit.path);
  if (busy !== undefined)
    throw invalidArgument(
      call,
      'Invalid Input: the org unit holds child org units, users, or devices.',
    );
  transaction(domain.db, () => {
    saveBeforeImage(
      domain.db,
      'orgunits',
      unit.org_unit_id,
      call.now,
      searchDelayMs(call),
    );
    domain.db
      .prepare('DELETE FROM orgunits WHERE org_unit_id = ?')
      .run(unit.org_unit_id);
  });
  return { status: 204 };
}
```
