# Admin SDK Mock Milestone 3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Google groups to the mock: seeded structural and class-section groups, and the `groups`, `groups.aliases`, and `members` methods with lagging lists, nested membership, cycle checks, and the nesting and ready delays.

**Architecture:** Three new tables hold groups, group aliases, and direct memberships. All three lag through the milestone 2 before-images. Nested membership is never stored. A recursive query follows group-to-group links on each read, and a link counts only after the nesting delay. Derived-membership pages read a reached-group set and scan the `(email, member_id)` index in order, so a 320,000-member expansion pages without a sort.

**Tech Stack:** Node 24 (type stripping, `node:test`, `node:sqlite`), TypeScript (type checks only), ESLint, Prettier. Compatibility tests use `@googleapis/admin` and `google-auth-library`. No new dependencies.

**Spec:** `docs/specs/2026-10-10-milestone-3-groups-design.md`. Its parents are `docs/specs/2026-10-09-admin-sdk-mock-design.md` and `docs/specs/2026-10-09-milestone-2-writes-design.md`. Read all three before any task.

## Global Constraints

- Repository: `~/projects/admin-sdk-mock`. Work on branch `codex/milestone-3` from `main`. Never add files to `~/projects/campus-commander`.
- Node runs `.ts` files directly through type stripping. Never use `enum`, `namespace`, parameter properties, decorators, or `import x = require()`.
- Relative imports end in `.ts`. Type-only imports use `import type`.
- Add no runtime or development dependency.
- Directory errors use `directoryError()` from `src/http/errors.ts` or the helpers in `src/directory/common.ts`. Never build an error envelope by hand.
- Group, alias, and member routes require a super administrator subject.
- Group and alias reads accept `SCOPES.group`. Member reads accept `SCOPES.member`. Group and alias writes accept `WRITE_SCOPES.group`. Member writes accept `WRITE_SCOPES.member`.
- Emails are stored in lower case. Keys match emails without regard to case. IDs match exactly.
- A write stamps its before-images with `call.now`. The search delay in milliseconds is `searchDelayMs(call)`. A delay of 0 saves no before-images.
- The time scale divides the search delay, the nesting delay, and the ready delay. `nestingDelayMs(call)` and `groupReadyMs(call)` return scaled milliseconds.
- A group link is followable when `added_at <= call.now - nestingDelayMs(call)`. Cycle checks follow every link.
- Lists read lagged relations: `listedTable(call, 'groups' | 'group_aliases' | 'group_members' | 'users')`. `get`, `hasMember`, writes, and write responses read live tables.
- Write handlers never set `items` in their response. The write latency model charges `perDeviceWriteMs` per item. Writes that remove membership rows set `scanned` to the row count.
- Write responses without a response schema return `204` with an empty body.
- Documentation prose follows the repository writing rules: active voice, no hedging words, no semicolons, no contractions, at most 20 words per instruction sentence.
- Every commit message ends with a `Co-Authored-By` trailer that names the model that wrote the commit, for example `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`.
- Every task ends with `npm test`, `npm run lint`, `npm run typecheck`, and `npm run format:check` passing.

  type GroupRow,
} from '../store/group-lookup.ts';

import { domainOf, notFound } from './common.ts';

const EMAIL = /^[^@\s]+@[^@\s]+\.[^@\s]+$/;

/** Find a group that requests may see by email, alias, or ID. Unknown and hidden groups answer 404. */
export function visibleGroup(call: Call, key: string): GroupRow {
  const domain = domainOf(call);
  const group = findGroup(domain.db, key);
  const hidden = call.overrides.hidden(domain.customerId);
  if (!group || hidden.has(group.id) || hidden.has(group.email))
    throw notFound('groupKey');
  return group;
}

/** What a member key names. `OUTSIDE` is an address on a domain outside the customer. */
export type MemberTarget =
  | { type: 'USER'; id: string; email: string; user: UserRow }
  | { type: 'GROUP'; id: string; email: string; group: GroupRow }
  | { type: 'CUSTOMER'; id: string; email: '' }
  | { type: 'OUTSIDE'; id: string; email: string };

/** Resolve a member key. Unknown keys, and hidden users and groups, resolve to undefined. */
export function resolveMemberTarget(
  call: Call,
  key: string,
): MemberTarget | undefined {
  const domain = domainOf(call);
  const hidden = call.overrides.hidden(domain.customerId);
  const user = findUser(domain.db, key);
  if (user && !hidden.has(user.id) && !hidden.has(user.primary_email))
    return { type: 'USER', id: user.id, email: user.primary_email, user };
  const group = findGroup(domain.db, key);
  if (group && !hidden.has(group.id) && !hidden.has(group.email))
    return { type: 'GROUP', id: group.id, email: group.email, group };
  if (key === domain.customerId)
    return { type: 'CUSTOMER', id: domain.customerId, email: '' };
  if (EMAIL.test(key) && domainKind(domain.db, key) === 'outside') {
    const email = key.toLowerCase();
    return { type: 'OUTSIDE', id: outsideMemberId(email), email };
  }
  return undefined;
}

/** The nesting delay of the call's domain in scaled milliseconds. */
export function nestingDelayMs(call: Call): number {
  return (
    (domainOf(call).record.settings.writes.nestingDelaySeconds * 1000) /
    call.scale
  );
}

/** The ready delay of the call's domain in scaled milliseconds. */
export function groupReadyMs(call: Call): number {
  return (
    (domainOf(call).record.settings.writes.groupReadySeconds * 1000) /
    call.scale
  );
}

/** Pick `count` distinct items in random order. */
export function sample<T>(random: Random, items: readonly T[], count: number): T[] {
  const pool = [...items];
  const n = Math.min(count, pool.length);
  for (let i = 0; i < n; i++) {
    const j = random.int(i, pool.length - 1);
    [pool[i], pool[j]] = [pool[j], pool[i]];
  }
  return pool.slice(0, n);
}

/** Insert groups and memberships, and keep every address and ID unique. */
function groupWriter(ctx: SeedContext, random: Random) {
  const { db, seededAt, spec } = ctx;
  const taken = new Set<string>(
    rows<{ address: string }>(
      db
        .prepare(
          'SELECT primary_email AS address FROM users UNION ALL SELECT alias FROM user_aliases',
        )
        .all(),
    ).map((entry) => entry.address),
  );
  const ids = new Set<string>();
  const groups: SeededGroup[] = [];
  let memberships = 0;
  const insertGroup = db.prepare(
    'INSERT INTO groups (id, email, name, description, admin_created, created_at) VALUES (?, ?, ?, ?, ?, ?)',
  );
  const insertMember = db.prepare(
    'INSERT INTO group_members (group_id, member_id, email, type, role, delivery, added_at) VALUES (?, ?, ?, ?, ?, ?, ?)',
  );
  const timing = random.fork('times');
  const delivery = random.fork('delivery');

  /** Reserve a unique address. A taken local part gains `-2`, `-3`, and so on. */
  const address = (local: string, domain: string): string => {
    let candidate = `${local}@${domain}`;
    for (let n = 2; taken.has(candidate); n++) candidate = `${local}-${n}@${domain}`;
    taken.add(candidate);
    return candidate;
  };

  const create = (input: {
    local: string;
    domain: string;
    name: string;
    description: string;
    kind: GroupKind;
    adminCreated?: boolean;
  }): SeededGroup => {
    const email = address(input.local, input.domain);
    const id = newGroupId(createRandom(`group/${spec.seed}/${email}`), (candidate) =>
      ids.has(candidate),
    );
    ids.add(id);
    const createdAt = seededAt - timing.int(30, 3000) * DAY_MS;
    insertGroup.run(
      id,
      email,
      input.name,
      input.description,
      input.adminCreated === false ? 0 : 1,
      createdAt,
    );
    const group: SeededGroup = {
      id,
      email,
      createdAt,
      kind: input.kind,
      members: 0,
      outside: false,
    };
    groups.push(group);
    return group;
  };

  /** Add one direct member. Membership times fall within 20 days of the group's creation. */
  const add = (
    group: SeededGroup,
    target: MemberTargetSeed,
    role: MemberRole = 'MEMBER',
    outside = false,
  ): void => {
    insertMember.run(
      group.id,
      target.id,
      target.email,
      target.type,
      role,
      delivery.weighted(DELIVERY),
      group.createdAt + timing.int(0, 20) * DAY_MS,
    );
    group.members++;
    if (outside) group.outside = true;
    memberships++;
  };

  return { groups, address, create, add, memberships: () => memberships };
}

export type GroupWriter = ReturnType<typeof groupWriter>;

/** The student OU that holds `ou`, or null when `ou` sits outside every student OU. */
function gradeOu(ou: string, grades: ReadonlyMap<string, unknown>): string | null {
  for (
    let path = ou;
    path.length > 1;
    path = path.slice(0, path.lastIndexOf('/')) || '/'
  )
    if (grades.has(path)) return path;
  return null;
}

/** Insert every seeded group. Runs after users. The caller owns the transaction. */
export function seedGroups(ctx: SeedContext): GroupSeedResult {
  const { spec, layout, db } = ctx;
  const random = createRandom(`groups/${spec.seed}`);
  const writer = groupWriter(ctx, random.fork('writer'));
  const users = rows<SeedUser>(
    db
      .prepare(
        'SELECT id, primary_email AS email, org_unit_path AS ou, role, school, family_name AS family FROM users ORDER BY ord',
      )
      .all(),
  );

  const grades = new Map<string, true>();
  for (const school of layout.schools)
    for (const grade of school.grades) grades.set(school.studentOus[grade], true);
  const studentsByOu = new Map<string, SeedUser[]>();
  const staffBySchool: SeedUser[][] = layout.schools.map(() => []);
  const districtStaff: SeedUser[] = [];
  const admins: SeedUser[] = [];
  for (const user of users) {
    if (user.role === 'student') {
      const ou = gradeOu(user.ou, grades);
      if (ou) studentsByOu.set(ou, [...(studentsByOu.get(ou) ?? []), user]);
    } else if (user.role === 'admin') {
      admins.push(user);
      districtStaff.push(user);
    } else if (user.school === null) districtStaff.push(user);
    else staffBySchool[user.school].push(user);
  }

  const schools: SchoolGroups[] = layout.schools.map((school) => {
    const pick = random.fork(`school/${school.index}`);
    const base = groupSlug(school.name);
    const staff = staffBySchool[school.index];
    const staffGroup = writer.create({
      local: `${base}-staff`,
      domain: spec.primaryDomain,
      name: `${school.name} Staff`,
      description: `All staff at ${school.name}`,
      kind: 'staff',
    });
    const owner = staff.length > 0 ? pick.int(0, staff.length - 1) : -1;
    const managers = new Set<number>();
    const managerCount = Math.min(pick.int(1, 2), staff.length - 1);
    while (managers.size < managerCount) {
      const n = pick.int(0, staff.length - 1);
      if (n !== owner) managers.add(n);
    }
    staff.forEach((user, n) =>
      writer.add(
        staffGroup,
        asUser(user),
        n === owner ? 'OWNER' : managers.has(n) ? 'MANAGER' : 'MEMBER',
      ),
    );

    const gradeGroups: GradeGroups[] = [];
    for (const grade of school.grades) {
      const ou = school.studentOus[grade];
      const students = studentsByOu.get(ou) ?? [];
      if (students.length === 0) continue;
      const leaf = ou.slice(ou.lastIndexOf('/') + 1);
      const group = writer.create({
        local: `${base}-${groupSlug(leaf)}`,
        domain: spec.studentDomain,
        name: `${school.name} ${leaf}`,
        description: `Students in ${leaf} at ${school.name}`,
        kind: 'grade',
      });
      for (const student of students) writer.add(group, asUser(student));
      gradeGroups.push({ group, grade, students });
    }

    const studentsGroup = writer.create({
      local: `${base}-students`,
      domain: spec.studentDomain,
      name: `${school.name} Students`,
      description: `All students at ${school.name}`,
      kind: 'students',
    });
    for (const { group } of gradeGroups) writer.add(studentsGroup, asGroup(group));
    const everyone = writer.create({
      local: `${base}-all`,
      domain: spec.primaryDomain,
      name: `${school.name} Everyone`,
      description: `Staff and students at ${school.name}`,
      kind: 'school',
    });
    writer.add(everyone, asGroup(staffGroup));
    writer.add(everyone, asGroup(studentsGroup));

    const schoolStudents = gradeGroups.flatMap((entry) => entry.students);
    const clubs: SeededGroup[] = [];
    for (let n = pick.int(1, 3); n > 0; n--) {
      const club = pick.pick(CLUBS);
      const group = writer.create({
        local: `${groupSlug(club)}-${base}`,
        domain: spec.studentDomain,
        name: `${school.name} ${club}`,
        description: `${club} at ${school.name}`,
        kind: 'club',
        adminCreated: false,
      });
      sample(pick, staff, pick.int(1, 4)).forEach((user, k) =>
        writer.add(
          group,
          asUser(user),
          k === 0 ? 'OWNER' : k === 1 ? 'MANAGER' : 'MEMBER',
        ),
      );
      for (const student of sample(pick, schoolStudents, pick.int(8, 30)))
        writer.add(group, asUser(student));
      clubs.push(group);
    }
    return {
      school,
      base,
      staff,
      staffGroup,
      owner: owner >= 0 ? staff[owner] : null,
      grades: gradeGroups,
      studentsGroup,
      clubs,
    };
  });

  const primary = spec.primaryDomain;
  const allStaff = writer.create({
    local: 'all-staff',
    domain: primary,
    name: 'All Staff',
    description: `Every staff member of ${spec.districtName}`,
    kind: 'district',
  });
  for (const user of districtStaff) writer.add(allStaff, asUser(user));
  const regions = new Map<string, SeededGroup>();
  for (const entry of schools) {
    const region = entry.school.region;
    if (region === null) {
      writer.add(allStaff, asGroup(entry.staffGroup));
      continue;
    }
    let group = regions.get(region);
    if (!group) {
      group = writer.create({
        local: `${groupSlug(region)}-staff`,
        domain: primary,
        name: `${region} Staff`,
        description: `Staff of the schools in ${region}`,
        kind: 'region',
      });
      regions.set(region, group);
      writer.add(allStaff, asGroup(group));
    }
    writer.add(group, asGroup(entry.staffGroup));
  }
  const allStudents = writer.create({
    local: 'all-students',
    domain: primary,
    name: 'All Students',
    description: `Every student of ${spec.districtName}`,
    kind: 'district',
  });
  for (const entry of schools) writer.add(allStudents, asGroup(entry.studentsGroup));
  const everyone = writer.create({
    local: 'everyone',
    domain: primary,
    name: 'Everyone',
    description: `Every user of ${spec.districtName}`,
    kind: 'district',
  });
  writer.add(everyone, { id: spec.customerId, email: '', type: 'CUSTOMER' });
  const tech = writer.create({
    local: 'tech-team',
    domain: primary,
    name: 'Technology Team',
    description: 'District technology administrators',
    kind: 'district',
  });
  admins.forEach((user, n) =>
    writer.add(tech, asUser(user), n === 0 ? 'OWNER' : 'MEMBER'),
  );
  const principals = writer.create({
    local: 'principals',
    domain: primary,
    name: 'Principals',
    description: 'The lead of each school',
    kind: 'district',
  });
  for (const entry of schools)
    if (entry.owner) writer.add(principals, asUser(entry.owner));

  return { groups: writer.groups.length, memberships: writer.memberships() };
}

```ts
import type { Random } from '../random.ts';
import { outsideMemberId } from '../store/group-lookup.ts';
import { emailLocal, type SeedContext } from './context.ts';
import { coursesFor, OUTSIDE_DOMAINS } from './courses.ts';
import {
  asGroup,
  asUser,
  groupSlug,
  sample,
  type GroupWriter,
  type MemberTargetSeed,
  type SchoolGroups,
  type SeededGroup,
  type SeedUser,
} from './groups.ts';
import { FIRST_NAMES, LAST_NAMES } from './names.ts';

export interface ScenarioInput {
  ctx: SeedContext;
  random: Random;
  writer: GroupWriter;
  schools: SchoolGroups[];
  admins: SeedUser[];
  districtStaff: SeedUser[];
}

/** Sections hold about 25 students. */
const SECTION_SIZE = 25;

/** A source of unique outside addresses, such as `ann.lee42@homemail.test`. */
function outsideAddresses(random: Random): () => MemberTargetSeed {
  const used = new Set<string>();
  return () => {
    for (;;) {
      const email = `${emailLocal(random.pick(FIRST_NAMES))}.${emailLocal(random.pick(LAST_NAMES))}${random.int(1, 99)}@${random.pick(OUTSIDE_DOMAINS)}`;
      if (used.has(email)) continue;
      used.add(email);
      return { id: outsideMemberId(email), email, type: 'USER' };
    }
  };
}

function seedSections({ ctx, random, writer, schools, districtStaff }: ScenarioInput): void {
  const perStudent = ctx.spec.rates.sectionsPerStudent;
  if (perStudent === 0) return;
  const pick = random.fork('sections');
  for (const entry of schools) {
    const teachers = entry.staff.length > 0 ? entry.staff : districtStaff;
    for (const { grade, students } of entry.grades) {
      const count = Math.ceil((students.length * perStudent) / SECTION_SIZE);
      const courses = coursesFor(grade);
      const sections: SeededGroup[] = [];
      for (let k = 0; k < count; k++) {
        const course = courses[k % courses.length];
        const period = (Math.floor(k / courses.length) % 8) + 1;
        const teacher = pick.pick(teachers);
        const group = writer.create({
          local: `${groupSlug(course)}-p${period}-${groupSlug(teacher.family)}`,
          domain: ctx.spec.studentDomain,
          name: `${course} – Period ${period} (${teacher.family})`,
          description: `${course} at ${entry.school.name}`,
          kind: 'section',
        });
        writer.add(group, asUser(teacher), 'OWNER');
        sections.push(group);
      }
      const joins = Math.min(perStudent, count);
      for (const student of students)
        for (const section of sample(pick, sections, joins))
          writer.add(section, asUser(student));
    }
  }
}

function seedOutsideGroups(
  { ctx, random, writer, schools, admins }: ScenarioInput,
  outside: () => MemberTargetSeed,
): void {
  const pick = random.fork('parents');
  for (const entry of schools) {
    const group = writer.create({
      local: `${entry.base}-parents`,
      domain: ctx.spec.primaryDomain,
      name: `${entry.school.name} Parent Organization`,
      description: `Parents and staff of ${entry.school.name}`,
      kind: 'parents',
    });
    if (entry.owner) writer.add(group, asUser(entry.owner), 'OWNER');
    for (let n = pick.int(3, 12); n > 0; n--)
      writer.add(group, outside(), 'MEMBER', true);
  }
  const board = writer.create({
    local: 'board',
    domain: ctx.spec.primaryDomain,
    name: 'School Board',
    description: `The board of ${ctx.spec.districtName}`,
    kind: 'board',
  });
  writer.add(board, asUser(admins[0]), 'OWNER');
  for (let n = pick.int(5, 9); n > 0; n--) writer.add(board, outside(), 'MEMBER', true);
}

function seedChains({ ctx, random, writer, schools, districtStaff }: ScenarioInput): void {
  const pick = random.fork('chains');
  const staff = [...districtStaff, ...schools.flatMap((entry) => entry.staff)];
  for (let chain = 1; chain <= ctx.spec.rates.deepGroupChains; chain++) {
    const levels = pick.int(8, 10);
    let parent: SeededGroup | null = null;
    for (let level = 1; level <= levels; level++) {
      const group = writer.create({
        local: `committee-${chain}-level-${level}`,
        domain: ctx.spec.primaryDomain,
        name: `District Committee ${chain} Level ${level}`,
        description: `Level ${level} of ${levels} in committee chain ${chain}`,
        kind: 'chain',
      });
      if (parent) writer.add(parent, asGroup(group));
      parent = group;
    }
    for (const user of sample(pick, staff, 3)) writer.add(parent!, asUser(user));
  }
}

/** Empty sections make up `emptyGroups` of all groups: E = N × r / (1 − r). */
function seedEmptyGroups({ ctx, random, writer, schools }: ScenarioInput): void {
  const rate = ctx.spec.rates.emptyGroups;
  const count = Math.round((writer.groups.length * rate) / (1 - rate));
  const pick = random.fork('empty');
  for (let n = 0; n < count; n++) {
    const entry = pick.pick(schools);
    const course = pick.pick(coursesFor(pick.pick(entry.school.grades)));
    const period = pick.int(1, 8);
    writer.create({
      local: `${groupSlug(course)}-p${period}-unassigned`,
      domain: ctx.spec.studentDomain,
      name: `${course} – Period ${period} (Unassigned)`,
      description: `An unscheduled section at ${entry.school.name}`,
      kind: 'empty',
    });
  }
}

/** Clubs and sections receive one outside guest each until the share is met. */
function seedOutsideGuests(
  { ctx, random, writer }: ScenarioInput,
  outside: () => MemberTargetSeed,
): void {
  const target = Math.round(writer.groups.length * ctx.spec.rates.externalMemberGroups);
  const have = writer.groups.filter((group) => group.outside).length;
  const candidates = writer.groups.filter(
    (group) => !group.outside && (group.kind === 'club' || group.kind === 'section'),
  );
  for (const group of sample(random.fork('guests'), candidates, target - have))
    writer.add(group, outside(), 'MEMBER', true);
}

function seedAliases({ ctx, random, writer }: ScenarioInput): void {
  const insert = ctx.db.prepare(
    'INSERT INTO group_aliases (alias, group_id) VALUES (?, ?)',
  );
  const count = Math.round(writer.groups.length * ctx.spec.rates.groupAliases);
  for (const group of sample(random.fork('aliases'), writer.groups, count)) {
    const local = group.email.slice(0, group.email.indexOf('@'));
    insert.run(writer.address(`${local}-list`, ctx.spec.primaryDomain), group.id);
  }
}

/** Seed the scenario families. Their order fixes every share computation. */
export function seedGroupScenarios(input: ScenarioInput): void {
  const outside = outsideAddresses(input.random.fork('outside'));
  seedSections(input);
  seedOutsideGroups(input, outside);
  seedChains(input);
  seedEmptyGroups(input);
  seedOutsideGuests(input, outside);
  seedAliases(input);
}
```

`sample()` returns an empty list for a count of 0 or less, so `seedOutsideGuests` adds nothing when parent groups and the board already meet the share.

```ts
import { emptyFilter, escapeLike, type SqlFilter } from '../store/sql.ts';
import { QueryError } from './device-query.ts';

const CLAUSE =
  /([A-Za-z]+)(=|:)('(?:[^'\\]|\\.)*'|\S*)|('(?:[^'\\]|\\.)*'|\S+)/g;

const unquote = (value: string): string =>
  value.length >= 2 && value.startsWith("'") && value.endsWith("'")
    ? value.slice(1, -1).replace(/\\(.)/g, '$1')
    : value;

/**
 * Translate the documented group search into SQL over the `groups` columns.
 * `email` and `name` take `=` or `:prefix*`. `memberKey` takes `=` and stands alone.
 */
export function parseGroupQuery(query: string): {
  filter: SqlFilter;
  memberKey: string | null;
} {
  const filter = emptyFilter();
  let memberKey: string | null = null;
  let named = false;
  for (const match of query.matchAll(CLAUSE)) {
    if (match[1] === undefined)
      throw new QueryError(`Unsupported query: ${match[4]}`);
    const field = match[1];
    const value = unquote(match[3]);
    if (field === 'memberKey') {
      if (match[2] !== '=') throw new QueryError('memberKey accepts only =.');
      memberKey = value;
      continue;
    }
    if (field !== 'email' && field !== 'name')
      throw new QueryError(`Unsupported query field: ${field}`);
    named = true;
    if (match[2] === '=') {
      filter.where.push(field === 'email' ? 'email = lower(?)' : 'lower(name) = lower(?)');
      filter.params.push(value);
    } else if (value.endsWith('*')) {
      filter.where.push(`lower(${field}) LIKE ? ESCAPE '\\'`);
      filter.params.push(`${escapeLike(value.slice(0, -1).toLowerCase())}%`);
    } else throw new QueryError(`${field} accepts only = or :prefix*.`);
  }
  if (memberKey !== null && named)
    throw new QueryError('memberKey cannot be combined with email or name.');
  return { filter, memberKey };
}
```

import { parseGroupQuery } from './group-query.ts';

const SORT_ORDER = { ascending: 'ASC', descending: 'DESC' } as const;
const ORDER_BY = { email: 'email' } as const;

/** Render a group with its aliases in order and its direct member count. */
export function groupResource(
  group: GroupRow,
  aliases: readonly string[],
  count: number,
): Record<string, unknown> {
  return withEtag({
    kind: 'admin#directory#group',
    id: group.id,
    email: group.email,
    name: group.name,
    description: group.description,
    directMembersCount: String(count),
    adminCreated: group.admin_created === 1,
    ...(aliases.length > 0 ? { aliases: [...aliases] } : {}),
  });
}

/** Direct member counts and aliases of several groups, read from the named relations. */
export function groupDetails(
  db: DatabaseSync,
  ids: readonly string[],
  memberTable = 'group_members',
  aliasTable = 'group_aliases',
): { counts: Map<string, number>; aliases: Map<string, string[]> } {
  const counts = new Map<string, number>();
  const aliases = new Map<string, string[]>();
  if (ids.length === 0) return { counts, aliases };
  const marks = ids.map(() => '?').join(', ');
  for (const entry of rows<{ group_id: string; n: number }>(
    db
      .prepare(
        `SELECT group_id, count(*) AS n FROM ${memberTable} WHERE group_id IN (${marks}) GROUP BY group_id`,
      )
      .all(...ids),
  ))
    counts.set(entry.group_id, Number(entry.n));
  for (const entry of rows<{ alias: string; group_id: string }>(
    db
      .prepare(
        `SELECT alias, group_id FROM ${aliasTable} WHERE group_id IN (${marks}) ORDER BY alias`,
      )
      .all(...ids),
  ))
    aliases.set(entry.group_id, [...(aliases.get(entry.group_id) ?? []), entry.alias]);
  return { counts, aliases };
}

export function getGroup(call: Call): ApiResponse {
  const domain = domainOf(call);
  const group = visibleGroup(call, call.params.groupKey);
  const details = groupDetails(domain.db, [group.id]);
  return {
    status: 200,
    items: 1,
    body: groupResource(
      group,
      details.aliases.get(group.id) ?? [],
      details.counts.get(group.id) ?? 0,
    ),
  };
}

/** `groups.list`. Lists lag writes by the search delay. */
export function listGroups(call: Call): ApiResponse {
  const domain = domainOf(call);
  const query = call.request.query;
  const customer = query.get('customer');
  const domainName = query.get('domain')?.toLowerCase();
  const userKey = query.get('userKey');
  if ((!customer && !domainName && !userKey) || (customer && userKey))
    throw invalidArgument(call, 'Bad Request', 'badRequest');
  const table = listedTable(call, 'groups');
  const members = listedTable(call, 'group_members');
  const aliasTable = listedTable(call, 'group_aliases');
  const filter = emptyFilter();
  if (customer) resolveCustomer(call, customer);
  if (domainName) {
    const known = row(
      domain.db
        .prepare(
          'SELECT 1 FROM domains WHERE domain_name = ? AND alias_of IS NULL',
        )
        .get(domainName),
    );
    if (!known) throw new ApiError(notAuthorized('directory'));
    filter.where.push('substr(email, ?) = ?');
    filter.params.push(-(domainName.length + 1), `@${domainName}`);
  }
  const directMember = (target: { email: string; id: string }) => {
    filter.where.push(
      `id IN (SELECT group_id FROM ${members} WHERE email = ? AND member_id = ?)`,
    );
    filter.params.push(target.email, target.id);
  };
  if (userKey) {
    // Ruling: a userKey that names no user or group of the customer returns 400.
    const target = resolveMemberTarget(call, userKey);
    if (!target || (target.type !== 'USER' && target.type !== 'GROUP'))
      throw invalidArgument(call, 'Invalid Input: userKey');
    directMember(target);
  }
  const maxResults = intParam(call, 'maxResults', {
    fallback: 200,
    min: 1,
    max: 200,
  });
  enumParam(call, 'orderBy', ORDER_BY, 'email');
  const descending = enumParam(call, 'sortOrder', SORT_ORDER, 'ASC') === 'DESC';
  const text = query.get('query');
  if (text) {
    let parsed: ReturnType<typeof parseGroupQuery>;
    try {
      parsed = parseGroupQuery(text);
    } catch (error) {
      if (error instanceof QueryError)
        throw invalidArgument(call, `Invalid Input: ${error.message}`);
      throw error;
    }
    filter.where.push(...parsed.filter.where);
    filter.params.push(...parsed.filter.params);
    if (parsed.memberKey !== null) {
      // Ruling: a memberKey that names nothing returns no groups.
      const target = resolveMemberTarget(call, parsed.memberKey);
      if (target) directMember(target);
      else filter.where.push('0 = 1');
    }
  }
  const hidden = [...call.overrides.hidden(domain.customerId)];
  if (hidden.length > 0) {
    const marks = hidden.map(() => '?').join(', ');
    filter.where.push(`id NOT IN (${marks}) AND email NOT IN (${marks})`);
    filter.params.push(...hidden, ...hidden);
  }
  const fp = fingerprint(call.request.path, query, domain.record.seededAt);
  const pageToken = query.get('pageToken');
  if (pageToken) {
    let cursor: (string | number)[];
    try {
      cursor = decodePageToken(pageToken, fp, 2);
    } catch {
      throw invalidArgument(call, 'Invalid Input: pageToken');
    }
    filter.where.push(`(email, id) ${descending ? '<' : '>'} (?, ?)`);
    filter.params.push(cursor[0], cursor[1]);
  }
  const direction = descending ? 'DESC' : 'ASC';
  const built = listedSelect(domain.db, 'groups', table, {
    extra: 'email AS sort_key',
    where: filter.where,
    params: filter.params,
    tail: `ORDER BY sort_key ${direction}, id ${direction} LIMIT ?`,
  });
  const found = rows<GroupRow & { sort_key: string }>(
    domain.db.prepare(built.sql).all(...built.params, maxResults + 1),
  );
  const page = found.slice(0, maxResults);
  const details = groupDetails(
    domain.db,
    page.map((group) => group.id),
    members,
    aliasTable,
  );
  const last = page.at(-1);
  const nextPageToken =
    found.length > maxResults && last
      ? encodePageToken(fp, [last.email, last.id])
      : undefined;
  const groups = page.map((group) =>
    groupResource(
      group,
      details.aliases.get(group.id) ?? [],
      details.counts.get(group.id) ?? 0,
    ),
  );
  const total = text
    ? Number(
        row<{ n: number }>(domain.db.prepare('SELECT count(*) AS n FROM groups').get())
          ?.n,
      )
    : 0;
  return {
    status: 200,
    items: page.length,
    scanned: total,
    body: withEtag({
      kind: 'admin#directory#groups',
      ...(groups.length > 0 ? { groups } : {}),
      ...(nextPageToken ? { nextPageToken } : {}),
    }),
  };
}

export const MAX_DESCRIPTION = 4096;

/** Read an address on one of the customer's own domains. Alias domains do not count. */
export function groupAddress(call: Call, value: unknown, field: string): string {
  if (typeof value !== 'string' || !EMAIL.test(value.trim()))
    throw invalidArgument(call, `Invalid Input: ${field}`);
  const email = value.trim().toLowerCase();
  if (domainKind(domainOf(call).db, email) !== 'own')
    throw invalidArgument(call, `Invalid Input: ${field}`);
  return email;
}

/** Render a group after a write, from the live tables. */
export function writtenGroup(call: Call, id: string): Record<string, unknown> {
  const domain = domainOf(call);
  const group = row<GroupRow>(
    domain.db.prepare('SELECT * FROM groups WHERE id = ?').get(id),
  )!;
  const details = groupDetails(domain.db, [id]);
  return groupResource(group, details.aliases.get(id) ?? [], details.counts.get(id) ?? 0);
}

const localPart = (email: string) => email.slice(0, email.indexOf('@'));

/** `groups.insert`. Ruling: `name` defaults to the local part of the email. */
export function insertGroup(call: Call): ApiResponse {
  const domain = domainOf(call);
  const body = jsonBody(call);
  const email = groupAddress(call, body.email, 'email');
  const name = stringField(call, body, 'name') || localPart(email);
  const description = stringField(call, body, 'description', MAX_DESCRIPTION) ?? '';
  if (addressInUse(domain.db, email)) throw duplicate();
  const taken = domain.db.prepare('SELECT 1 FROM groups WHERE id = ?');
  const id = newGroupId(
    createRandom(`group/${domain.spec.seed}/${email}/${call.now}`),
    (candidate) => taken.get(candidate) !== undefined,
  );
  transaction(domain.db, () => {
    saveBeforeImage(domain.db, 'groups', id, call.now, searchDelayMs(call));
    domain.db
      .prepare(
        'INSERT INTO groups (id, email, name, description, admin_created, created_at) VALUES (?, ?, ?, ?, 1, ?)',
      )
      .run(id, email, name, description, call.now);
  });
  return { status: 200, body: writtenGroup(call, id) };
}

/** `groups.update` and `groups.patch`. Both change only the fields present in the body. */
export function updateGroup(call: Call): ApiResponse {
  const domain = domainOf(call);
  const group = visibleGroup(call, call.params.groupKey);
  const body = jsonBody(call);
  const sets: string[] = [];
  const params: string[] = [];
  const name = stringField(call, body, 'name');
  if (name !== undefined) {
    sets.push('name = ?');
    params.push(name || localPart(group.email));
  }
  const description = stringField(call, body, 'description', MAX_DESCRIPTION);
  if (description !== undefined) {
    sets.push('description = ?');
    params.push(description ?? '');
  }
  let rename: string | undefined;
  if (body.email !== undefined && body.email !== null) {
    const email = groupAddress(call, body.email, 'email');
    if (email !== group.email) {
      if (addressInUse(domain.db, email, { groupId: group.id })) throw duplicate();
      rename = email;
    }
  }
  if (sets.length > 0 || rename) {
    const delay = searchDelayMs(call);
    transaction(domain.db, () => {
      saveBeforeImage(domain.db, 'groups', group.id, call.now, delay);
      if (sets.length > 0)
        domain.db
          .prepare(`UPDATE groups SET ${sets.join(', ')} WHERE id = ?`)
          .run(...params, group.id);
      if (!rename) return;
      // Ruling: the old address stays as an alias, as a user rename keeps it.
      saveBeforeImage(domain.db, 'group_aliases', rename, call.now, delay);
      saveBeforeImage(domain.db, 'group_aliases', group.email, call.now, delay);
      domain.db
        .prepare('DELETE FROM group_aliases WHERE alias = ? AND group_id = ?')
        .run(rename, group.id);
      domain.db.prepare('UPDATE groups SET email = ? WHERE id = ?').run(rename, group.id);
      domain.db
        .prepare('INSERT INTO group_aliases (alias, group_id) VALUES (?, ?)')
        .run(group.email, group.id);
      const asMember = "email = ? AND member_id = ? AND type = 'GROUP'";
      saveBeforeImagesWhere(
        domain.db,
        'group_members',
        asMember,
        [group.email, group.id],
        call.now,
        delay,
      );
      domain.db
        .prepare(`UPDATE group_members SET email = ? WHERE ${asMember}`)
        .run(rename, group.email, group.id);
    });
  }
  return { status: 200, body: writtenGroup(call, group.id) };
}

/** `groups.delete`. Removes the group, its aliases, its members, and its membership in other groups. */
export function deleteGroup(call: Call): ApiResponse {
  const domain = domainOf(call);
  const group = visibleGroup(call, call.params.groupKey);
  const delay = searchDelayMs(call);
  const asMember = "email = ? AND member_id = ? AND type = 'GROUP'";
  let removed = 0;
  transaction(domain.db, () => {
    saveBeforeImage(domain.db, 'groups', group.id, call.now, delay);
    saveBeforeImagesWhere(domain.db, 'group_aliases', 'group_id = ?', [group.id], call.now, delay);
    saveBeforeImagesWhere(domain.db, 'group_members', 'group_id = ?', [group.id], call.now, delay);
    saveBeforeImagesWhere(
      domain.db,
      'group_members',
      asMember,
      [group.email, group.id],
      call.now,
      delay,
    );
    domain.db.prepare('DELETE FROM group_aliases WHERE group_id = ?').run(group.id);
    removed += Number(
      domain.db.prepare('DELETE FROM group_members WHERE group_id = ?').run(group.id).changes,
    );
    removed += Number(
      domain.db
        .prepare(`DELETE FROM group_members WHERE ${asMember}`)
        .run(group.email, group.id).changes,
    );
    domain.db.prepare('DELETE FROM groups WHERE id = ?').run(group.id);
  });
  return { status: 204, scanned: removed };
}

import type { DatabaseSync } from 'node:sqlite';
import { row, rows } from '../store/sqlite.ts';

/**
 * Every group reached from `groupId`, including `groupId`, through group links
 * added at or before `followableAt`. `table` is `group_members` or its listed view.
 * UNION, not UNION ALL, keeps the walk finite even if a cycle ever existed.
 */
export function reachedGroups(
  db: DatabaseSync,
  table: string,
  groupId: string,
  followableAt: number,
): string[] {
  return rows<{ id: string }>(
    db
      .prepare(
        `WITH RECURSIVE reach(id) AS (SELECT ? UNION SELECT m.member_id FROM ${table} m JOIN reach r ON m.group_id = r.id WHERE m.type = 'GROUP' AND m.added_at <= ?) SELECT id FROM reach`,
      )
      .all(groupId, followableAt),
  ).map((entry) => entry.id);
}

/** True when `to` is `from` or lies below it through any live group link. Cycle checks use this. */
export function reaches(db: DatabaseSync, from: string, to: string): boolean {
  return (
    row(
      db
        .prepare(
          "WITH RECURSIVE reach(id) AS (SELECT ? UNION SELECT m.member_id FROM group_members m JOIN reach r ON m.group_id = r.id WHERE m.type = 'GROUP') SELECT 1 FROM reach WHERE id = ? LIMIT 1",
        )
        .get(from, to),
    ) !== undefined
  );

import type { DatabaseSync } from 'node:sqlite';
import type { ApiResponse, Call } from '../http/types.ts';
import {
  ROLE_RANK,
  ROLES,
  type GroupRow,
  type MemberRole,
  type MemberRow,
} from '../store/group-lookup.ts';
import { listedSelect } from '../store/lag.ts';
import {
  decodePageToken,
  encodePageToken,
  fingerprint,
} from '../store/page-token.ts';
import { row, rows } from '../store/sqlite.ts';
import {
  boolParam,
  domainOf,
  intParam,
  invalidArgument,
  listedTable,
  notFound,
  withEtag,
} from './common.ts';
import {
  nestingDelayMs,
  resolveMemberTarget,
  visibleGroup,
} from './group-common.ts';
import { reachedGroups } from './group-graph.ts';

/** Derived pages scan the email index once the reached groups hold more rows than this. */
export const EMAIL_SCAN_ROWS = 5000;

type Listed = Pick<MemberRow, 'member_id' | 'email' | 'type' | 'role'> & {
  delivery?: string;
};

/** Render one membership. List entries omit `delivery_settings`, as Google documents. */
export function memberResource(
  member: Listed,
  suspended: boolean,
  withDelivery: boolean,
): Record<string, unknown> {
  return withEtag({
    kind: 'admin#directory#member',
    id: member.member_id,
    ...(member.type === 'CUSTOMER' ? {} : { email: member.email }),
    role: member.role,
    type: member.type,
    // Ruling: SUSPENDED for a suspended user, ACTIVE for every other member.
    status: suspended ? 'SUSPENDED' : 'ACTIVE',
    ...(withDelivery && member.delivery ? { delivery_settings: member.delivery } : {}),
  });
}

/** The IDs among `ids` that are suspended users in the named users relation. */
export function suspendedUsers(
  db: DatabaseSync,
  table: string,
  ids: readonly string[],
): Set<string> {
  if (ids.length === 0) return new Set();
  return new Set(
    rows<{ id: string }>(
      db
        .prepare(
          `SELECT id FROM ${table} WHERE suspended = 1 AND id IN (${ids.map(() => '?').join(', ')})`,
        )
        .all(...ids),
    ).map((entry) => entry.id),
  );
}

/** Render a membership from the live tables, with its delivery setting. */
export function liveMember(db: DatabaseSync, member: MemberRow): Record<string, unknown> {
  const suspended =
    member.type === 'USER' && suspendedUsers(db, 'users', [member.member_id]).size > 0;
  return memberResource(member, suspended, true);
}

/** Find a direct membership by member key. Ruling: a nested member answers 404. */
export function directMember(call: Call, group: GroupRow, key: string): MemberRow {
  const db = domainOf(call).db;
  const target = resolveMemberTarget(call, key);
  const found = target
    ? row<MemberRow>(
        db
          .prepare(
            'SELECT * FROM group_members WHERE group_id = ? AND email = ? AND member_id = ?',
          )
          .get(group.id, target.email, target.id),
      )
    : row<MemberRow>(
        db
          .prepare('SELECT * FROM group_members WHERE group_id = ? AND member_id = ?')
          .get(group.id, key),
      );
  if (!found) throw notFound('memberKey');
  return found;
}

function rolesParam(call: Call): MemberRole[] | null {
  const raw = call.request.query.get('roles');
  if (raw === null || raw === '') return null;
  const roles = raw.split(',').map((role) => role.trim());
  if (!roles.every((role) => (ROLES as readonly string[]).includes(role)))
    throw invalidArgument(call, 'Invalid Input: roles');
  return roles as MemberRole[];
}

/**
 * The derived-membership page query. Parameters, in order: the reached group IDs
 * as a JSON array, the cursor email and member ID when `cursor` is set, one rank
 * per role filter, and the row limit. On the live table, large sets scan the
 * `(email, member_id)` index in order, so GROUP BY and ORDER BY need no sort.
 */
export function derivedQuery(
  table: string,
  total: number,
  options: { cursor: boolean; roles: number; hidden?: number },
): string {
  const hint =
    table !== 'group_members'
      ? ''
      : total > EMAIL_SCAN_ROWS
        ? ' INDEXED BY group_members_email'
        : ' INDEXED BY group_members_order';
  const hidden = options.hidden
    ? ` AND member_id NOT IN (${Array(options.hidden).fill('?').join(', ')})`
    : '';
  const having =
    options.roles > 0
      ? ` HAVING rank IN (${Array(options.roles).fill('?').join(', ')})`
      : '';
  return (
    `SELECT member_id, email, type, MIN(CASE role WHEN 'OWNER' THEN 0 WHEN 'MANAGER' THEN 1 ELSE 2 END) AS rank ` +
    `FROM ${table}${hint} WHERE group_id IN (SELECT value FROM json_each(?))` +
    (options.cursor ? ' AND (email, member_id) > (?, ?)' : '') +
    hidden +
    ` GROUP BY email, member_id${having} ORDER BY email, member_id LIMIT ?`
  );
}

/** `members.list`. Lists lag writes. Derived lists also wait for the nesting delay. */
export function listMembers(call: Call): ApiResponse {
  const domain = domainOf(call);
  const db = domain.db;
  const query = call.request.query;
  const group = visibleGroup(call, call.params.groupKey);
  const maxResults = intParam(call, 'maxResults', { fallback: 200, min: 1, max: 200 });
  const roles = rolesParam(call);
  const derived = boolParam(call, 'includeDerivedMembership');
  const table = listedTable(call, 'group_members');
  const users = listedTable(call, 'users');
  const hidden = [...call.overrides.hidden(domain.customerId)];
  const fp = fingerprint(call.request.path, query, domain.record.seededAt);
  const pageToken = query.get('pageToken');
  let cursor: (string | number)[] | null = null;
  if (pageToken) {
    try {
      cursor = decodePageToken(pageToken, fp, 2);
    } catch {
      throw invalidArgument(call, 'Invalid Input: pageToken');
    }
  }
  let found: Listed[];
  let scanned = 0;
  if (derived) {
    const reach = JSON.stringify(
      reachedGroups(db, table, group.id, call.now - nestingDelayMs(call)),
    );
    scanned = Number(
      row<{ n: number }>(
        db
          .prepare(
            `SELECT count(*) AS n FROM ${table} WHERE group_id IN (SELECT value FROM json_each(?))`,
          )
          .get(reach),
      )?.n ?? 0,
    );
    found = rows<{ member_id: string; email: string; type: Listed['type']; rank: number }>(
      db
        .prepare(
          derivedQuery(table, scanned, {
            cursor: cursor !== null,
            roles: roles?.length ?? 0,
            hidden: hidden.length,
          }),
        )
        .all(
          reach,
          ...(cursor ?? []),
          ...hidden,
          ...(roles ?? []).map((role) => ROLE_RANK[role]),
          maxResults + 1,
        ),
    ).map((entry) => ({ ...entry, role: ROLES[entry.rank] }));
  } else {
    const where = ['group_id = ?'];
    const params: (string | number)[] = [group.id];
    if (roles) {
      where.push(`role IN (${roles.map(() => '?').join(', ')})`);
      params.push(...roles);
    }
    if (cursor) {
      where.push('(email, member_id) > (?, ?)');
      params.push(cursor[0], cursor[1]);
    }
    if (hidden.length > 0) {
      where.push(`member_id NOT IN (${hidden.map(() => '?').join(', ')})`);
      params.push(...hidden);
    }
    const built = listedSelect(db, 'group_members', table, {
      extra: 'email AS sort_key',
      where,
      params,
      tail: 'ORDER BY sort_key, member_id LIMIT ?',
    });
    found = rows<MemberRow>(db.prepare(built.sql).all(...built.params, maxResults + 1));
  }
  const page = found.slice(0, maxResults);
  const last = page.at(-1);
  const nextPageToken =
    found.length > maxResults && last
      ? encodePageToken(fp, [last.email, last.member_id])
      : undefined;
  const suspended = suspendedUsers(
    db,
    users,
    page.filter((member) => member.type === 'USER').map((member) => member.member_id),
  );
  const members = page.map((member) =>
    memberResource(member, suspended.has(member.member_id), false),
  );
  return {
    status: 200,
    items: page.length,
    scanned,
    body: withEtag({
      kind: 'admin#directory#members',
      ...(members.length > 0 ? { members } : {}),
      ...(nextPageToken ? { nextPageToken } : {}),
    }),
  };
}

export function getMember(call: Call): ApiResponse {
  const domain = domainOf(call);
  const group = visibleGroup(call, call.params.groupKey);
  const member = directMember(call, group, call.params.memberKey);
  return { status: 200, items: 1, body: liveMember(domain.db, member) };
}

import { ApiError, directoryError } from '../http/errors.ts';
import type { ApiResponse, Call } from '../http/types.ts';
import {
  DELIVERIES,
  ROLES,
  type MemberRow,
  type MemberType,
} from '../store/group-lookup.ts';
import { saveBeforeImage } from '../store/lag.ts';
import { row, transaction } from '../store/sqlite.ts';
import {
  domainOf,
  invalidArgument,
  jsonBody,
  notFound,
  searchDelayMs,
} from './common.ts';
import {
  groupReadyMs,
  resolveMemberTarget,
  visibleGroup,
} from './group-common.ts';
import { reaches } from './group-graph.ts';
import { directMember, liveMember } from './members.ts';

/** Ruling: a duplicate membership answers 409 with this message. */
const memberExists = () =>
  new ApiError(directoryError(409, 'duplicate', 'Member already exists.'));

/** Read an optional enum field. Null or absent reads as undefined. */
function enumField<T extends string>(
  call: Call,
  body: Record<string, unknown>,
  name: string,
  values: readonly T[],
): T | undefined {
  const value = body[name];
  if (value === undefined || value === null) return undefined;
  if (typeof value !== 'string' || !(values as readonly string[]).includes(value))
    throw invalidArgument(call, `Invalid Input: ${name}`);
  return value as T;
}

const memberById = (call: Call, id: number) =>
  row<MemberRow>(domainOf(call).db.prepare('SELECT * FROM group_members WHERE id = ?').get(id))!;

/** `members.insert`. The type follows the key: user, group, customer, or outside address. */
export function insertMember(call: Call): ApiResponse {
  const domain = domainOf(call);
  const db = domain.db;
  const group = visibleGroup(call, call.params.groupKey);
  // Ruling: inserting into a group younger than the ready delay answers 404.
  if (call.now - group.created_at < groupReadyMs(call)) throw notFound('groupKey');
  const body = jsonBody(call);
  const key =
    typeof body.email === 'string' && body.email !== ''
      ? body.email
      : typeof body.id === 'string' && body.id !== ''
        ? body.id
        : undefined;
  if (key === undefined) throw invalidArgument(call, 'Invalid Input: memberKey');
  const role = enumField(call, body, 'role', ROLES) ?? 'MEMBER';
  const delivery = enumField(call, body, 'delivery_settings', DELIVERIES) ?? 'ALL_MAIL';
  // Ruling: an address on a customer domain that names nothing answers 404.
  const target = resolveMemberTarget(call, key);
  if (!target) throw notFound('memberKey');
  const type: MemberType =
    target.type === 'GROUP' ? 'GROUP' : target.type === 'CUSTOMER' ? 'CUSTOMER' : 'USER';
  if (
    row(
      db
        .prepare(
          'SELECT 1 FROM group_members WHERE group_id = ? AND email = ? AND member_id = ?',
        )
        .get(group.id, target.email, target.id),
    )
  )
    throw memberExists();
  // Ruling: a cycle answers 400. The check follows links still inside the nesting delay.
  if (type === 'GROUP' && reaches(db, target.id, group.id))
    throw invalidArgument(call, 'Cyclic memberships not allowed.');
  // A new ID past every live row and every waiting before-image never reuses a key.
  const id = Number(
    row<{ n: number }>(
      db
        .prepare(
          'SELECT MAX(COALESCE((SELECT MAX(id) FROM group_members), 0), COALESCE((SELECT MAX(id) FROM group_members_prior), 0)) + 1 AS n',
        )
        .get(),
    )!.n,
  );
  transaction(db, () => {
    saveBeforeImage(db, 'group_members', id, call.now, searchDelayMs(call));
    db.prepare(
      'INSERT INTO group_members (id, group_id, member_id, email, type, role, delivery, added_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?)',
    ).run(id, group.id, target.id, target.email, type, role, delivery, call.now);
  });
  return { status: 200, body: liveMember(db, memberById(call, id)) };
}

/** `members.update` and `members.patch`. Both change only `role` and `delivery_settings`. */
export function updateMember(call: Call): ApiResponse {
  const db = domainOf(call).db;
  const group = visibleGroup(call, call.params.groupKey);
  const member = directMember(call, group, call.params.memberKey);
  const body = jsonBody(call);
  const role = enumField(call, body, 'role', ROLES);
  const delivery = enumField(call, body, 'delivery_settings', DELIVERIES);
  if (role !== undefined || delivery !== undefined)
    transaction(db, () => {
      saveBeforeImage(db, 'group_members', member.id, call.now, searchDelayMs(call));
      db.prepare('UPDATE group_members SET role = ?, delivery = ? WHERE id = ?').run(
        role ?? member.role,
        delivery ?? member.delivery,
        member.id,
      );
    });
  return { status: 200, body: liveMember(db, memberById(call, member.id)) };
}

/** `members.delete`. Removes one direct membership. */
export function deleteMember(call: Call): ApiResponse {
  const db = domainOf(call).db;
  const group = visibleGroup(call, call.params.groupKey);
  const member = directMember(call, group, call.params.memberKey);
  transaction(db, () => {
    saveBeforeImage(db, 'group_members', member.id, call.now, searchDelayMs(call));
    db.prepare('DELETE FROM group_members WHERE id = ?').run(member.id);
  });
  return { status: 204 };
}

import type { ApiResponse, Call } from '../http/types.ts';
import { row } from '../store/sqlite.ts';
import { domainOf, invalidArgument, notFound } from './common.ts';
import {
  nestingDelayMs,
  resolveMemberTarget,
  visibleGroup,
} from './group-common.ts';
import { reachedGroups } from './group-graph.ts';

const domainPart = (email: string) => email.slice(email.lastIndexOf('@') + 1);
const IN_REACH = 'group_id IN (SELECT value FROM json_each(?))';

/**
 * `members.hasMember`. Direct membership answers on any domain. A nested check
 * needs the user and the group on one email domain, as Google documents.
 */
export function hasMember(call: Call): ApiResponse {
  const db = domainOf(call).db;
  const group = visibleGroup(call, call.params.groupKey);
  const target = resolveMemberTarget(call, call.params.memberKey);
  // Ruling: an unknown key answers 404. Groups, outside addresses, and the customer answer 400.
  if (!target) throw notFound('memberKey');
  if (target.type !== 'USER') throw invalidArgument(call, 'Invalid Input: memberKey');
  const direct = row(
    db
      .prepare(
        'SELECT 1 FROM group_members WHERE group_id = ? AND email = ? AND member_id = ?',
      )
      .get(group.id, target.email, target.id),
  );
  if (direct) return { status: 200, items: 1, body: { isMember: true } };
  if (domainPart(target.email) !== domainPart(group.email))
    throw invalidArgument(call, 'Invalid Input');
  const reach = JSON.stringify(
    reachedGroups(db, 'group_members', group.id, call.now - nestingDelayMs(call)),
  );
  const found =
    row(
      db
        .prepare(
          `SELECT 1 FROM group_members WHERE email = ? AND member_id = ? AND ${IN_REACH} LIMIT 1`,
        )
        .get(target.email, target.id, reach),
    ) ??
    row(
      db
        .prepare(
          `SELECT 1 FROM group_members WHERE email = '' AND type = 'CUSTOMER' AND ${IN_REACH} LIMIT 1`,
        )
        .get(reach),
    );
  return { status: 200, items: 1, body: { isMember: found !== undefined } };
}

/** End every pending ready delay and nesting delay. Returns the rows changed. */
export function endGroupDelays(
  db: DatabaseSync,
  cutoffs: { readyAfter: number; nestingAfter: number },

  const groups = db

    .prepare(
      "UPDATE group_members SET added_at = 0 WHERE type = 'GROUP' AND added_at > ?",
